_resampler_cache = {}
def preprocess_waveform_fixed(waveform, sr):
    if waveform.dim() == 2:
        waveform = waveform.squeeze(0)
    if sr != TARGET_SR:
        key = int(sr)
        if key not in _resampler_cache:
            _resampler_cache[key] = torchaudio.transforms.Resample(orig_freq=sr, new_freq=TARGET_SR)
        waveform = _resampler_cache[key](waveform)
    T = waveform.numel()
    if T < FIXED_SAMPLES:
        pad = torch.zeros(FIXED_SAMPLES - T, dtype=waveform.dtype)
        waveform = torch.cat([waveform, pad], dim=0)
    else:
        waveform = waveform[:FIXED_SAMPLES]
    return waveform

def preprocess_waveform_logmel_vec(waveform, sr):
    w = preprocess_waveform_fixed(waveform, sr)
    M = to_db(mel(w))  # [n_mels, frames]
    fT = M.shape[-1]
    if fT < FIXED_FRAMES:
        pad = torch.zeros(N_MELS, FIXED_FRAMES - fT, dtype=M.dtype)
        M = torch.cat([M, pad], dim=1)
    else:
        M = M[:, :FIXED_FRAMES]
    # per-sample norm
    M = (M - M.mean()) / (M.std() + 1e-5)
    return M.reshape(-1).float()

def collate_sc(batch):
    xs, ys = [], []
    for waveform, sr, raw_label, *_ in batch:
        x = preprocess_waveform_logmel_vec(waveform, sr)
        lab = map_to_pipe_label(raw_label) if USE_HF_PIPELINE_BASELINE else raw_label
        if lab not in label2id:
            lab = "_unknown_" if "_unknown_" in label2id else list(label2id.keys())[0]
        xs.append(x)
        ys.append(label2id[lab])
    X = torch.stack(xs, dim=0)
    Y = torch.tensor(ys, dtype=torch.long)
    return X, Y

def exp_schedule_safe(step, start_step, end_step, start_val, end_val):
    step = max(step, start_step)
    end_step = max(end_step, start_step + 1)
    t = (step - start_step) / (end_step - start_step)
    t = min(max(t, 0.0), 1.0)
    if start_val <= 0 or end_val <= 0:
        return start_val + (end_val - start_val) * t
    return start_val * ((end_val / start_val) ** t)

def ste_bernoulli_gate(p):
    hard = torch.bernoulli(p)
    return (hard - p).detach() + p

def estimate_macs(in_dim, h1, h2, out_dim, g1_ratio, g2_ratio):
    baseline = in_dim*h1 + h1*h2 + h2*out_dim
    gated = in_dim*h1 + (h1*g1_ratio)*h2 + (h2*g2_ratio)*out_dim
    return baseline, gated, (gated / max(baseline, 1.0))

class BaselineMLP(nn.Module):
    def __init__(self, in_dim, out_dim, h1, h2):
        super().__init__()
        self.fc1 = nn.Linear(in_dim, h1)
        self.fc2 = nn.Linear(h1, h2)
        self.fc3 = nn.Linear(h2, out_dim)
    def forward(self, x):
        h1 = F.relu(self.fc1(x))
        h2 = F.relu(self.fc2(h1))
        return self.fc3(h2)

class DynamicGateMLP(nn.Module):
    def __init__(self, in_dim, out_dim, cfg):
        super().__init__()
        self.cfg = cfg
        self.h1 = cfg["h1"]; self.h2 = cfg["h2"]
        self.fc1 = nn.Linear(in_dim, self.h1)
        self.fc2 = nn.Linear(self.h1, self.h2)
        self.fc3 = nn.Linear(self.h2, out_dim)

        self.gate_threshold = float(cfg["gate_threshold_eval"])
        mode = cfg["gating_mode"]

        if mode == "static":
            self.gate_logit1 = nn.Parameter(torch.zeros(self.h1))
            self.gate_logit2 = nn.Parameter(torch.zeros(self.h2))
            with torch.no_grad():
                self.gate_logit1.add_(torch.randn_like(self.gate_logit1) * cfg["init_noise"])
                self.gate_logit2.add_(torch.randn_like(self.gate_logit2) * cfg["init_noise"])
        elif mode == "samplewise":
            gh = int(cfg["gate_hidden"])
            self.gnet1 = nn.Sequential(nn.Linear(self.h1, gh), nn.ReLU(), nn.Linear(gh, self.h1))
            self.gnet2 = nn.Sequential(nn.Linear(self.h2, gh), nn.ReLU(), nn.Linear(gh, self.h2))
            self.gate_bias1 = nn.Parameter(torch.zeros(self.h1))
            self.gate_bias2 = nn.Parameter(torch.zeros(self.h2))
            with torch.no_grad():
                self.gate_bias1.add_(torch.randn_like(self.gate_bias1) * cfg["init_noise"])
                self.gate_bias2.add_(torch.randn_like(self.gate_bias2) * cfg["init_noise"])
        else:
            raise ValueError("gating_mode must be 'static' or 'samplewise'")

    def init_gate_logits(self, p0=0.7):
        p0 = float(p0)
        p0 = min(max(p0, 1e-4), 1-1e-4)
        logit = math.log(p0/(1-p0))
        with torch.no_grad():
            if self.cfg["gating_mode"] == "static":
                self.gate_logit1.fill_(logit)
                self.gate_logit2.fill_(logit)
            else:
                self.gate_bias1.fill_(logit)
                self.gate_bias2.fill_(logit)

    def forward(self, x, tau=1.0, return_gates=False):
        h1 = F.relu(self.fc1(x))

        if self.cfg["gating_mode"] == "static":
            p1 = torch.sigmoid(self.gate_logit1 / max(tau, 1e-6))
            p1_b = p1.unsqueeze(0).expand(h1.size(0), -1)
        else:
            logit1 = self.gnet1(h1) + self.gate_bias1
            p1_b = torch.sigmoid(logit1 / max(tau, 1e-6))
            p1 = p1_b.mean(dim=0)

        if self.training:
            g1_b = ste_bernoulli_gate(p1_b)
        else:
            g1_b = (p1_b > self.gate_threshold).float()

        h1g = h1 * g1_b
        h2 = F.relu(self.fc2(h1g))

        if self.cfg["gating_mode"] == "static":
            p2 = torch.sigmoid(self.gate_logit2 / max(tau, 1e-6))
            p2_b = p2.unsqueeze(0).expand(h2.size(0), -1)
        else:
            logit2 = self.gnet2(h2) + self.gate_bias2
            p2_b = torch.sigmoid(logit2 / max(tau, 1e-6))
            p2 = p2_b.mean(dim=0)

        if self.training:
            g2_b = ste_bernoulli_gate(p2_b)
        else:
            g2_b = (p2_b > self.gate_threshold).float()

        h2g = h2 * g2_b
        logits = self.fc3(h2g)

        if return_gates:
            return logits, (p1_b, p2_b), (g1_b.detach(), g2_b.detach())
        return logits

def gate_regularizer(p1_b, p2_b, cfg, lam: float, target_keep=None):
    keep1 = p1_b.mean()
    keep2 = p2_b.mean()

    tgt = float(cfg["target_keep"] if target_keep is None else target_keep)


    if cfg["reg_type"] == "linear":
        over1 = F.relu(keep1 - tgt)
        over2 = F.relu(keep2 - tgt)
        loss_gate = lam * (over1 + over2)
    elif cfg["reg_type"] == "mse":
        loss_gate = lam * ((keep1 - tgt)**2 + (keep2 - tgt)**2)
    else:
        raise ValueError("reg_type must be 'linear' or 'mse'")

    return loss_gate, float(keep1.item()), float(keep2.item())

@torch.no_grad()
def evaluate_model(model, loader, is_gate=False):
    model.eval()
    correct, total = 0, 0
    p1_list, p2_list, g1_list, g2_list = [], [], [], []

    for x, y in loader:
        x = x.to(device, non_blocking=True)
        y = y.to(device, non_blocking=True)
        if is_gate:
            logits, (p1_b, p2_b), (g1_b, g2_b) = model(x, tau=1.0, return_gates=True)
            p1_list.append(float(p1_b.mean().item()))
            p2_list.append(float(p2_b.mean().item()))
            g1_list.append(float(g1_b.float().mean().item()))
            g2_list.append(float(g2_b.float().mean().item()))
        else:
            logits = model(x)

        pred = logits.argmax(dim=1)
        correct += (pred == y).sum().item()
        total += y.numel()

    acc = correct / max(total, 1)
    stats = None
    if is_gate:
        stats = {
            "p1": float(np.mean(p1_list)), "p2": float(np.mean(p2_list)),
            "g1": float(np.mean(g1_list)), "g2": float(np.mean(g2_list)),
        }
    return acc, stats

@torch.no_grad()
def evaluate_pipe_on_dataset(ds, max_items=2000, batch_size=32):
    if pipe is None:
        return None

    correct, total = 0, 0
    inputs, gts = [], []

    n = min(len(ds), int(max_items)) if max_items is not None else len(ds)
    for i in range(n):
        waveform, sr, raw_label, *_ = ds[i]
        w = preprocess_waveform_fixed(waveform, sr)
        lab = map_to_pipe_label(raw_label)
        gt = label2id.get(lab, label2id.get("_unknown_", 0))

        inputs.append({"array": w.numpy().astype(np.float32), "sampling_rate": TARGET_SR})
        gts.append(gt)

        if len(inputs) == batch_size:
            out = pipe(inputs, top_k=1)
            for o, gt_id in zip(out, gts):
                pred_lab = o["label"] if isinstance(o, dict) else o[0]["label"]
                pred_id = label2id.get(pred_lab, label2id.get("_unknown_", 0))
                correct += int(pred_id == gt_id)
                total += 1
            inputs, gts = [], []

    if len(inputs) > 0:
        out = pipe(inputs, top_k=1)
        for o, gt_id in zip(out, gts):
            pred_lab = o["label"] if isinstance(o, dict) else o[0]["label"]
            pred_id = label2id.get(pred_lab, label2id.get("_unknown_", 0))
            correct += int(pred_id == gt_id)
            total += 1

    return correct / max(total, 1)

    t0 = time.time()
    best_val = 0.0

    for ep in range(1, EPOCHS_BASELINE + 1):
        model.train()
        for x, y in train_loader:
            x = x.to(device, non_blocking=True)
            y = y.to(device, non_blocking=True)
            with torch.cuda.amp.autocast(enabled=(AMP and device == "cuda")):
                logits = model(x)
                loss = F.cross_entropy(logits, y)

            opt.zero_grad(set_to_none=True)
            scaler.scale(loss).backward()
            scaler.step(opt)
            scaler.update()

        val_acc, _ = evaluate_model(model, val_loader, is_gate=False)
        best_val = max(best_val, val_acc)
        elapsed = time.time() - t0
        print(f"[Baseline] ep {ep:02d}/{EPOCHS_BASELINE} val_acc={val_acc:.4f} best={best_val:.4f} time={elapsed:.1f}s")

    test_acc, _ = evaluate_model(model, test_loader, is_gate=False)
    baseline_macs, _, _ = estimate_macs(in_dim, CFG["h1"], CFG["h2"], num_classes, 1.0, 1.0)

    print("\n=== FINAL TEST (Baseline) ===")
    print("test_acc:", test_acc)
    print("MACs (baseline):", f"{baseline_macs:,}")
    return model, test_acc

# 10) Train: DynamicGate
# -------------------------
def train_dynamicgate():
    model = DynamicGateMLP(in_dim, num_classes, CFG).to(device)
    model.init_gate_logits(p0=CFG["init_p0"])
    opt = torch.optim.AdamW(model.parameters(), lr=LR, weight_decay=WEIGHT_DECAY)
    scaler = torch.cuda.amp.GradScaler(enabled=(AMP and device == "cuda"))

    global_step = 0
    steps_total = len(train_loader) * EPOCHS_GATE
    t0 = time.time()

# ---- (ADD) epoch phase boundaries ----
    e1 = max(1, int(round(GATE_PHASE1_FRAC * EPOCHS_GATE)))
    e2 = max(e1 + 1, int(round(GATE_PHASE2_FRAC * EPOCHS_GATE)))

    def phase_params(ep):
        # returns: (target_keep, eval_threshold, lambda_multiplier)
        if ep <= e1:
            return GATE_TARGET_KEEP_S1, GATE_THR_EVAL_S1, GATE_LAM_MUL_S1
        elif ep <= e2:
            return GATE_TARGET_KEEP_S2, GATE_THR_EVAL_S2, GATE_LAM_MUL_S2
        else:
            return GATE_TARGET_KEEP_S3, GATE_THR_EVAL_S3, GATE_LAM_MUL_S3




    if USE_HF_PIPELINE_BASELINE:
        pipe_val0 = evaluate_pipe_on_dataset(val_ds, max_items=PIPE_MAX_ITEMS_VAL, batch_size=PIPE_BATCH)
        print(f"[pipe] val_acc (mapped, n={PIPE_MAX_ITEMS_VAL}) = {pipe_val0:.4f}")

    for ep in range(1, EPOCHS_GATE + 1):
  # ---- (ADD) 3-phase 설정 ----
        target_keep_ep, thr_eval_ep, lam_mul = phase_params(ep)

 # eval/infer hard-threshold를 epoch에 따라 변화 (초반 많이 열고, 종반 닫힘)
        model.gate_threshold = float(thr_eval_ep)


        model.train()
        for x, y in train_loader:
            global_step += 1
            x = x.to(device, non_blocking=True)
            y = y.to(device, non_blocking=True)

            tau = exp_schedule_safe(global_step, 1, steps_total, CFG["tau_start"], CFG["tau_end"])
            lam = 0.0 if global_step < CFG["warmup_steps"] else exp_schedule_safe(
                global_step, CFG["warmup_steps"], steps_total, CFG["lambda_start"], CFG["lambda_end"]
            )
            lam = lam * lam_mul  # ✅ phase별로 sparsity 강도 조절

            with torch.cuda.amp.autocast(enabled=(AMP and device == "cuda")):
                logits, (p1_b, p2_b), _ = model(x, tau=tau, return_gates=True)
                loss_task = F.cross_entropy(logits, y)
                loss_gate, keep1, keep2 = gate_regularizer(
                   # p1_b, p2_b, CFG, lam
                    p1_b, p2_b, CFG, lam, target_keep=target_keep_ep  # ✅ phase별 target_keep
                )
                loss = loss_task + loss_gate

            opt.zero_grad(set_to_none=True)
            scaler.scale(loss).backward()
            scaler.step(opt)
            scaler.update()

        val_acc, gs = evaluate_model(model, val_loader, is_gate=True)
        base_macs, gated_macs, rel = estimate_macs(in_dim, CFG["h1"], CFG["h2"], num_classes, gs["g1"], gs["g2"])
        elapsed = time.time() - t0

# ---- (CHG) 로그에 phase 정보 포함 ----
        msg = (
            f"[Gate] ep {ep:02d}/{EPOCHS_GATE} val_acc={val_acc:.4f} "
            f"(phase_keep={target_keep_ep:.2f}, thr_eval={thr_eval_ep:.2f}, lam_mul={lam_mul:.1f}) "
            f"val_acc={val_acc:.4f} "
            f"p1={gs['p1']:.3f}, g1={gs['g1']:.3f} | "
            f"p2={gs['p2']:.3f}, g2={gs['g2']:.3f} | "
            f"RelMAC={rel:.3f} time={elapsed:.1f}s"
        )

        if USE_HF_PIPELINE_BASELINE and (ep % PIPE_EVAL_EVERY_EPOCH == 0):
            pipe_val = evaluate_pipe_on_dataset(val_ds, max_items=PIPE_MAX_ITEMS_VAL, batch_size=PIPE_BATCH)
            msg += f" | pipe_val_acc={pipe_val:.4f}"

        print(msg)

    # ---- (ADD) 종반 threshold 확정 후 최종 평가(닫힌 상태로 test) ----
    model.gate_threshold = float(GATE_THR_EVAL_S3)

    test_acc, gs = evaluate_model(model, test_loader, is_gate=True)
    base_macs, gated_macs, rel = estimate_macs(in_dim, CFG["h1"], CFG["h2"], num_classes, gs["g1"], gs["g2"])

    print("\n=== FINAL TEST (DynamicGate) ===")
    print("test_acc:", test_acc)
    print("Gate stats:", gs)
    print(f"MACs baseline={base_macs:,} | gated~={int(gated_macs):,} | rel={rel:.3f}")

    if USE_HF_PIPELINE_BASELINE:
        pipe_test = evaluate_pipe_on_dataset(test_ds, max_items=PIPE_MAX_ITEMS_TEST, batch_size=PIPE_BATCH)
        print(f"[pipe] test_acc (mapped, n={PIPE_MAX_ITEMS_TEST}) = {pipe_test:.4f}")

    return model, test_acc, gs
