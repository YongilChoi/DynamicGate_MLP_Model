# -------------------- 모델 정의 --------------------
class MLP(nn.Module):
    def __init__(self, input_dim=input_dim, hidden_dim=hidden, output_dim=output_dim):
        super().__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.fc2 = nn.Linear(hidden_dim, output_dim)
    def forward(self, x):
        x = x.view(x.size(0), -1)
        x = F.relu(self.fc1(x))
        return self.fc2(x)

class DropoutMLP(nn.Module):
    def __init__(self, input_dim=input_dim, hidden_dim=hidden, output_dim=output_dim, p=0.5):
        super().__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.drop = nn.Dropout(p)
        self.fc2 = nn.Linear(hidden_dim, output_dim)
    def forward(self, x):
        x = x.view(x.size(0), -1)
        x = F.relu(self.fc1(x))
        x = self.drop(x)
        return self.fc2(x)

# ✅ GateLayer: last_probs/last_hard 저장 (FLOPs 추정에 사용)
class GateLayer(nn.Module):
    def __init__(self, size, threshold=0.5, temp=1.0, init_logit=0.0):
        super().__init__()
        self.gate_logit = nn.Parameter(torch.ones(size) * init_logit)
        self.threshold = threshold
        self.temp = temp
        # last stats (buffer 느낌으로 저장)
        self.last_probs = None
        self.last_hard = None

    def forward(self, x):
        probs = torch.sigmoid(self.gate_logit / self.temp)     # (size,)
        hard = (probs > self.threshold).float()                # (size,)
        gates = hard + (probs - hard).detach()                 # STE

        # ✅ 저장 (detach해서 그래프 밖으로)
        self.last_probs = probs.detach()
        self.last_hard  = hard.detach()

        return x * gates, probs, hard



class DynamicGateMLP(nn.Module):
    def __init__(self, input_dim=input_dim, hidden_dim=hidden, output_dim=output_dim,
                 threshold=0.5, temp=1.0):
        super().__init__()
        self.gate_in  = GateLayer(input_dim, threshold=threshold, temp=temp, init_logit=0.0)
        self.fc1      = nn.Linear(input_dim, hidden_dim)
        self.gate_hid = GateLayer(hidden_dim, threshold=threshold, temp=temp, init_logit=0.0)
        self.fc2      = nn.Linear(hidden_dim, output_dim)

    def forward(self, x):
        x = x.view(x.size(0), -1)
        x, pin, hin = self.gate_in(x)
        x = self.fc1(x)
        x, ph, hh = self.gate_hid(x)
        x = F.relu(x)
        logits = self.fc2(x)
        # probs/hard 둘 다 리턴 (지표를 일관되게 만들기 위해)
        return logits, ph, hh

# -------------------- 학습/평가 함수 --------------------
def _macs_linear_dense(in_dim, out_dim):
    return int(in_dim * out_dim)

def _macs_elemwise(dim):
    return int(dim)

def _macs_linear_sparse_from_mask(mask: torch.Tensor):
    # mask: (out, in) with 0/1
    return int(mask.sum().item())

def flops_breakdown(model, input_dim, hidden_dim, output_dim, assume_skip=False, include_gate_mul=True):
    """
    assume_skip=False: 현재 코드(as-implemented) 기준 (dense matmul을 그대로 한다)
    assume_skip=True : 게이트/마스크를 이용해 실제로 matmul을 스킵했다고 가정(theoretical)
    반환: dict(layer->MACs) + total_MACs
    """

    # 1) gate open ratios (없으면 1.0로 처리)
    open_in = 1.0
    open_hid = 1.0

    if hasattr(model, "gate_in") and getattr(model.gate_in, "last_hard", None) is not None:
        open_in = float(model.gate_in.last_hard.mean().item())

    if hasattr(model, "gate_hid") and getattr(model.gate_hid, "last_hard", None) is not None:
        open_hid = float(model.gate_hid.last_hard.mean().item())

    # 2) fc1 형태: dense(fc1) vs RigL sparse(fc1_rigl)
    has_dense_fc1 = hasattr(model, "fc1") and isinstance(getattr(model, "fc1"), nn.Linear)
    has_rigl_fc1  = hasattr(model, "fc1_rigl")  # SparseLinearRigL wrapper

    breakdown = {}

    # gate_in elementwise multiply (실제로는 x*gates 연산이 존재)
    if include_gate_mul and hasattr(model, "gate_in"):
        breakdown["gate_in_mul"] = _macs_elemwise(input_dim)
    else:
        breakdown["gate_in_mul"] = 0

    # fc1
    if has_dense_fc1:
        if assume_skip:
            # 게이트로 입력 feature를 실제로 제외하고 matmul 했다고 가정
            eff_in = max(1, int(round(input_dim * open_in)))
            breakdown["fc1"] = _macs_linear_dense(eff_in, hidden_dim)
        else:
            # 현재 구현: gate로 0 만들어도 dense matmul은 그대로 수행
            breakdown["fc1"] = _macs_linear_dense(input_dim, hidden_dim)

    elif has_rigl_fc1:
        # RigL sparse weights 기반
        mask = model.fc1_rigl.mask  # (out, in)
        nnz = _macs_linear_sparse_from_mask(mask)

        if assume_skip:
            # input gate까지 활용해 "활성 input에 걸린 연결만" 계산한다고 가정
            # 정확 계산은 어렵고, 빠른 근사: nnz * open_in (독립 가정)
            breakdown["fc1_rigl"] = int(round(nnz * open_in))
        else:
            # 현재 구현: fc1_rigl forward는 이미 mask가 곱해진 dense F.linear라서
            # "실제 연산"은 framework상 dense일 수 있음(진짜 sparse matmul이 아님).
            # 여기서는 '구조적으로 남은 연결 수(nnz)'를 MACs로 보고 싶다면 nnz로,
            # 실제 커널 실행 비용을 보수적으로 잡고 싶다면 input_dim*hidden_dim로 잡을 수도 있음.
            # 👉 여기서는 연구 관점(구조적 MACs)로 nnz를 반환.
            breakdown["fc1_rigl"] = nnz

    # gate_hid elementwise multiply
    if include_gate_mul and hasattr(model, "gate_hid"):
        breakdown["gate_hid_mul"] = _macs_elemwise(hidden_dim)
    else:
        breakdown["gate_hid_mul"] = 0

    # fc2
    if hasattr(model, "fc2") and isinstance(getattr(model, "fc2"), nn.Linear):
        if assume_skip:
            eff_h = max(1, int(round(hidden_dim * open_hid)))
            breakdown["fc2"] = _macs_linear_dense(eff_h, output_dim)
        else:
            breakdown["fc2"] = _macs_linear_dense(hidden_dim, output_dim)

    breakdown["total_MACs"] = sum(v for k, v in breakdown.items() if k != "total_MACs")

    # FLOPs로 보고 싶으면 보통 MACs*2 (mul+add)로 표기
    breakdown["total_FLOPs(2*MACs)"] = 2 * breakdown["total_MACs"]

    # 참고 정보도 같이
    breakdown["open_in(hard_mean)"] = open_in
    breakdown["open_hid(hard_mean)"] = open_hid

    return breakdown

def train_epoch(model, loader, opt, device, crit,
                epoch=0, gate_beta=0.0, target_sparsity=None, warmup_epochs=0,
                prune_mask=None):
    model.train(); total, correct = 0, 0

    # target open-rate
    target_open = None
    if target_sparsity is not None:
        target_open = 1.0 - float(target_sparsity)

    for x, y in loader:
        x, y = x.to(device, non_blocking=True), y.to(device, non_blocking=True)
        opt.zero_grad()

        out = model(x)

        if isinstance(out, tuple) and len(out) == 3:
            logits, probs, hard = out
            loss = crit(logits, y)

            # warmup 동안 gate loss를 꺼서 붕괴 방지
            if epoch >= warmup_epochs and gate_beta > 0.0:
                if target_open is None:
                    # 그냥 열림 최소화(붕괴 위험↑) 대신 완만하게
                    gate_loss = probs.mean()
                else:
                    # ✅ target open-rate에 맞추는 loss (붕괴 방지에 훨씬 안정적)
                   # gate_loss = (probs.mean() - target_open) ** 2
                    gate_loss = (probs.mean() - target_open)**2 + α * ReLU(min_open - probs.mean())
                loss = loss + gate_beta * gate_loss
        else:
            logits = out
            loss = crit(logits, y)

        loss.backward()
        opt.step()

        # ✅ pruning mask 유지: step 후 다시 0을 강제
        if prune_mask is not None:
            with torch.no_grad():
                model.fc1.weight.mul_(prune_mask)

        correct += (logits.argmax(1) == y).sum().item()
        total += y.size(0)

    return correct / total

@torch.no_grad()
def test_epoch(model, loader, device, crit):
    model.eval(); total, correct = 0, 0
    gate_probs = []
    gate_hard_ratio = []

    for x, y in loader:
        x, y = x.to(device, non_blocking=True), y.to(device, non_blocking=True)
        out = model(x)

        if isinstance(out, tuple) and len(out) == 3:
            logits, probs, hard = out
            gate_probs.append(probs.mean().item())
            gate_hard_ratio.append(hard.mean().item())
        else:
            logits = out

        correct += (logits.argmax(1) == y).sum().item()
        total += y.size(0)

    mean_probs = (sum(gate_probs)/len(gate_probs)) if gate_probs else None
    mean_hard  = (sum(gate_hard_ratio)/len(gate_hard_ratio)) if gate_hard_ratio else None
    return correct / total, mean_probs, mean_hard

# -------------------- FLOPs & Params 계산 --------------------
def compute_stats(model, active_ratio=None):
    params = sum(p.numel() for p in model.parameters() if p.requires_grad)

    if hasattr(model, 'fc1') and hasattr(model, 'fc2'):
        in_dim  = model.fc1.in_features
        hid_dim = model.fc1.out_features
        out_dim = model.fc2.out_features
        flops = in_dim * hid_dim + hid_dim * out_dim
    else:
        flops = 0

    # active_ratio는 "실제 활성 비율(=hard.mean 등)"로 넣는 것을 권장
    if active_ratio is not None:
        flops = int(flops * float(active_ratio))

    return params, flops

# -------------------- Pruning 함수 (mask 유지) --------------------
def apply_pruning_with_mask(model, ratio=0.3):
    with torch.no_grad():
        weight = model.fc1.weight.abs()
        thresh = torch.quantile(weight.view(-1), ratio)
        mask = (weight > thresh).float()
        model.fc1.weight.mul_(mask)
    return model, mask

# -------------------- Gate 시각화 --------------------
def show_gate_matrix(model, rows=16, cols=16):
    if not hasattr(model, "gate_hid"):
        print("This model has no gate_hid.")
        return

    g = model.gate_hid
    with torch.no_grad():
        probs = torch.sigmoid(g.gate_logit / g.temp).cpu().numpy()
        hard = (probs > g.threshold).astype(int)

    df_probs = pd.DataFrame(probs.reshape(1, -1))
    df_hard  = pd.DataFrame(hard.reshape(1, -1))

    print(f"\n[Hidden Gate] Gate probabilities (0~1):")
    try:
        from IPython.display import display
        display(df_probs)
    except:
        print(df_probs.head())

    print(f"\n[Hidden Gate] Gate hard values (0/1):")
    try:
        from IPython.display import display
        display(df_hard)
    except:
        print(df_hard.head())

    print(f"\n[Hidden Gate] Active gates: {df_hard.values.sum()} / {df_hard.size}")

    try:
        hard_matrix = hard.reshape(rows, cols)
        plt.figure(figsize=(5,4))
        plt.imshow(hard_matrix, cmap="Greens", aspect="auto")
        plt.colorbar(label="Gate (0=off, 1=on)")
        plt.title("DynamicGate-MLP Hidden Gate Status")
        plt.tight_layout()
        plt.show()
    except Exception as e:
        print("Cannot reshape to matrix:", e)
