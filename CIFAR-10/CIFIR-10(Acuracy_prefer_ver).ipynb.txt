@torch.no_grad()
def accuracy(model, loader):
    model.eval()
    correct, total = 0, 0
    for x,y in loader:
        x,y = x.to(device), y.to(device)
        x = x.view(x.size(0), -1)
        logits = model(x)
        pred = logits.argmax(dim=1)
        correct += (pred==y).sum().item()
        total += y.size(0)
    return 100.0 * correct / total

def train_simple(model, loader, epochs, lr=LR, desc="train", label_smoothing=0.1):
    model.to(device)
    opt = optim.AdamW(model.parameters(), lr=lr, weight_decay=WEIGHT_DECAY)
    criterion = nn.CrossEntropyLoss(label_smoothing=label_smoothing)

    # Cosine 스케줄 (정확도에 도움되는 경우 많음)
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)

    for ep in range(1, epochs+1):
        model.train()
        total_loss = 0.0
        for x,y in loader:
            x,y = x.to(device), y.to(device)
            x = x.view(x.size(0), -1)
            opt.zero_grad(set_to_none=True)
            logits = model(x)
            loss = criterion(logits, y)
            loss.backward()
            opt.step()
            total_loss += loss.item() * x.size(0)

        sched.step()
        print(f"[{desc}] epoch {ep:02d}/{epochs} - loss: {total_loss/len(loader.dataset):.4f} - lr:{sched.get_last_lr()[0]:.3g}")



# FLOPs(상대치) 계산: FC 레이어 MAC 기준
def fc_macs(in_features, out_features):  # bias 제외
    return int(in_features) * int(out_features)

def dense_mlp_flops():
    return (
        fc_macs(INPUT_DIM, HID1) +
        fc_macs(HID1, HID2) +
        fc_macs(HID2, NUM_CLASSES)
    )

BASELINE_FLOPS = dense_mlp_flops()

class MLP(nn.Module):
    def __init__(self, dropout_p=0.0):
        super().__init__()
        self.fc1 = nn.Linear(INPUT_DIM, HID1)
        self.fc2 = nn.Linear(HID1, HID2)
        self.fc3 = nn.Linear(HID2, NUM_CLASSES)
        self.do1 = nn.Dropout(dropout_p) if dropout_p>0 else nn.Identity()
        self.do2 = nn.Dropout(dropout_p) if dropout_p>0 else nn.Identity()
    def forward(self, x):
        x = F.relu(self.fc1(x))
        x = self.do1(x)
        x = F.relu(self.fc2(x))
        x = self.do2(x)
        x = self.fc3(x)
        return x

# 프루닝 후 FLOPs(상대) 추정: 남은 가중치 비율로 근사
def nonzero_ratio_linear(linear):
    w = linear.weight.detach()
    nz = (w != 0).sum().item()
    total = w.numel()
    return nz / total

r1 = nonzero_ratio_linear(pruned.fc1)
r2 = nonzero_ratio_linear(pruned.fc2)
r3 = nonzero_ratio_linear(pruned.fc3)

)

# 7) 4) Gate-MLP (하드 게이트 + STE)
# -----------------------------
class GateSTEFn(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, tau, thr):
        p = torch.sigmoid(logits / tau)
        g = (p > thr).float()
        ctx.save_for_backward(p, torch.tensor(float(tau), dtype=logits.dtype, device=logits.device))
        return g
    @staticmethod
    def backward(ctx, grad_out):
        p, tau = ctx.saved_tensors
        grad_logits = grad_out * (p * (1 - p)) / tau
        return grad_logits, None, None

class GatedLinear(nn.Module):
    def __init__(self, in_f, out_f, tau=2.0, thr=0.5, p0=0.8):
        super().__init__()
        self.fc = nn.Linear(in_f, out_f)
        z0 = torch.logit(torch.tensor(p0))
        self.logits = nn.Parameter(torch.full((out_f,), float(z0)))
        self.tau = tau
        self.thr = thr
    def forward(self, x, return_p=False):
        h = self.fc(x)
        g = GateSTEFn.apply(self.logits, self.tau, self.thr)  # (out_f,)
        y = h * g
        if return_p:
            p = torch.sigmoid(self.logits / self.tau)
            return y, p, g
        return y

class GateMLP(nn.Module):
    def __init__(self, tau=2.0, thr=0.5, p0=0.8):
        super().__init__()
        self.g1 = GatedLinear(INPUT_DIM, HID1, tau=tau, thr=thr, p0=p0)
        self.g2 = GatedLinear(HID1, HID2,    tau=tau, thr=thr, p0=p0)
        self.fc3 = nn.Linear(HID2, NUM_CLASSES)
    def forward(self, x, return_stats=False):
        y1, p1, g1 = self.g1(x, return_p=True)
        a1 = F.relu(y1)
        y2, p2, g2 = self.g2(a1, return_p=True)
        a2 = F.relu(y2)
        out = self.fc3(a2)
        if return_stats:
            return out, (p1, p2), (g1, g2)
        return out

def train_gate(model, loader, epochs, lr=LR,
               lambda_start=0.0, lambda_end=1e-4,
               warmup_epochs=10,
               tau_hi=2.0, tau_lo=1.5, tau_anneal_portion=0.7,
               target1=1.0, target2=1.0, target_w=0.0,  # 정확도 최우선이면 target_w=0
               label_smoothing=0.1):
    model.to(device)
    opt = optim.AdamW(model.parameters(), lr=lr, weight_decay=WEIGHT_DECAY)
    criterion = nn.CrossEntropyLoss(label_smoothing=label_smoothing)

    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)

    for ep in range(1, epochs+1):
        # tau anneal (너무 hard로 안 가게 tau_lo=1.5 권장)
        t = min(1.0, ep / max(1, int(epochs * tau_anneal_portion)))
        tau_now = tau_hi + (tau_lo - tau_hi) * t
        model.g1.tau = model.g2.tau = tau_now

        # lambda schedule (아주 약하게)
        if ep <= warmup_epochs:
            lambda_now = 0.0
        else:
            tt = (ep - warmup_epochs) / max(1, (epochs - warmup_epochs))
            lambda_now = lambda_start + (lambda_end - lambda_start) * tt

        model.train()
        total_loss = 0.0
        mean_p1s, mean_p2s = [], []

        for x,y in loader:
            x,y = x.to(device), y.to(device)
            x = x.view(x.size(0), -1)

            opt.zero_grad(set_to_none=True)
            logits, (p1,p2), _ = model(x, return_stats=True)

            task_loss = criterion(logits, y)

            # 정확도 최우선: sparsity 규제는 아주 약하게(또는 0)
            sparsity_loss = lambda_now * (p1.mean() + p2.mean())

            # 정확도 최우선: target 유도는 OFF (target_w=0이면 영향 없음)
            target_loss = target_w * ((p1.mean() - target1)**2 + (p2.mean() - target2)**2)

            loss = task_loss + sparsity_loss + target_loss
            loss.backward()
            torch.nn.utils.clip_grad_norm_([model.g1.logits, model.g2.logits], max_norm=1.0)
            opt.step()

            total_loss += loss.item() * x.size(0)
            mean_p1s.append(p1.mean().item()); mean_p2s.append(p2.mean().item())

        sched.step()
        print(f"[Gate-MLP] epoch {ep:02d}/{epochs} - loss: {total_loss/len(loader.dataset):.4f} "
              f"- tau:{tau_now:.2f} - lam:{lambda_now:.2g} "
              f"- mean(p1/p2): {np.mean(mean_p1s):.3f}/{np.mean(mean_p2s):.3f} - lr:{sched.get_last_lr()[0]:.3g}")


@torch.no_grad()
def eval_gate(model, loader, thr=0.5):
    model.eval()
    correct, total = 0, 0
    ps_all = []
    for x,y in loader:
        x,y = x.to(device), y.to(device)
        x = x.view(x.size(0), -1)
        logits, (p1,p2), _ = model(x, return_stats=True)
        pred = logits.argmax(dim=1)
        correct += (pred==y).sum().item()
        total += y.size(0)
        ps_all.append(torch.cat([p1, p2]).detach().cpu().numpy())

    acc = 100.0 * correct / total
    ps  = np.concatenate(ps_all, axis=0)

    # FLOPs 추정(하드마스크 기준): 레이어별 열린 뉴런 수로 근사
    p1 = torch.sigmoid(model.g1.logits / model.g1.tau).detach().cpu().numpy()
    p2 = torch.sigmoid(model.g2.logits / model.g2.tau).detach().cpu().numpy()
    g1_mask = (p1 > thr).astype(np.float32)
    g2_mask = (p2 > thr).astype(np.float32)

    on1 = g1_mask.sum()
    on2 = g2_mask.sum()

    flops = fc_macs(INPUT_DIM, on1) + fc_macs(on1, on2) + fc_macs(on2, NUM_CLASSES)
    flops_rel = flops / BASELINE_FLOPS

    open_rate1 = on1 / HID1
    open_rate2 = on2 / HID2

    return acc, flops_rel, ps, (open_rate1, open_rate2)
